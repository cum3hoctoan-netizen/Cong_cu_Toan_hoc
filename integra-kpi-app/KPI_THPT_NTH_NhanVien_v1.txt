        function updatePartAStatus(itemId, val) {
            let itemObj = null;
            PART_A_ITEMS.forEach(g => { g.items.forEach(i => { if (i.id === itemId) itemObj = i; }); });
            if (!itemObj) return;

            if (val === 'DAM_BAO') {
                userState.partAInputs[itemId].status = 'DAM_BAO';
                userState.partAInputs[itemId].score = itemObj.maxScore;
            } else {
                userState.partAInputs[itemId].status = 'KHONG_DAM_BAO';
                userState.partAInputs[itemId].score = 0;
            }

            // Update DOM element in place (NO full tbody re-render to prevent DOM destruction)
            const scoreInp = document.getElementById("partA_score_" + itemId);
            if (scoreInp) scoreInp.value = userState.partAInputs[itemId].score;

            recalculatePartATotals();
        }

        function updatePartAScore(itemId, val) {
            let itemObj = null;
            PART_A_ITEMS.forEach(g => { g.items.forEach(i => { if (i.id === itemId) itemObj = i; }); });
            if (!itemObj) return;

            let num = parseFloat(val);
            if (isNaN(num)) num = 0;
            if (num > itemObj.maxScore) num = itemObj.maxScore;
            if (num < 0) num = 0;

            userState.partAInputs[itemId].score = num;
            userState.partAInputs[itemId].status = (num === itemObj.maxScore) ? 'DAM_BAO' : 'KHONG_DAM_BAO';

            // Update status select in place (NO full tbody re-render)
            const statusSel = document.getElementById("partA_status_" + itemId);
            if (statusSel) statusSel.value = userState.partAInputs[itemId].status;

            recalculatePartATotals();
        }

        function recalculatePartATotals() {
            let sum1 = 0, sum2 = 0, sum3 = 0;
            PART_A_ITEMS[0].items.forEach(i => {
                sum1 += (userState.partAInputs[i.id] || { score: i.maxScore }).score;
            });
            PART_A_ITEMS[1].items.forEach(i => {
                sum2 += (userState.partAInputs[i.id] || { score: i.maxScore }).score;
            });
            PART_A_ITEMS[2].items.forEach(i => {
                sum3 += (userState.partAInputs[i.id] || { score: i.maxScore }).score;
            });

            if (document.getElementById("partA_grp_1_sum")) document.getElementById("partA_grp_1_sum").innerText = sum1.toFixed(2);
            if (document.getElementById("partA_grp_2_sum")) document.getElementById("partA_grp_2_sum").innerText = sum2.toFixed(2);
            if (document.getElementById("partA_grp_3_sum")) document.getElementById("partA_grp_3_sum").innerText = sum3.toFixed(2);

            const totalA = sum1 + sum2 + sum3;

            if (document.getElementById("score_partA1_disp")) document.getElementById("score_partA1_disp").innerText = sum1.toFixed(2) + '/18';
            if (document.getElementById("score_partA2_disp")) document.getElementById("score_partA2_disp").innerText = sum2.toFixed(2) + '/4';
            if (document.getElementById("score_partA3_disp")) document.getElementById("score_partA3_disp").innerText = sum3.toFixed(2) + '/8';
            if (document.getElementById("score_partA_total")) document.getElementById("score_partA_total").innerText = totalA.toFixed(2) + ' / 30đ';

            userState.scoreA1 = sum1;
            userState.scoreA2 = sum2;
            userState.scoreA3 = sum3;

            recalculateTotal();
        }

        function recalculateTotal() {
            const activeRoles = getActiveRoles();
            let totalA = 0;
            let totalB = 0;
            let countX = 0;
            let totalTasks = 0;
            let bonusFromAdHoc = 0;

            MASTER_DB.forEach(t => {
                if (t.Required_Roles.includes("REMOVE")) return;
                const hasRole = t.Required_Roles.some(r => activeRoles.includes(r) || r === "ALL_NV" || r === "ALL_GV" || r === "TU_CHON_HOAC_PHAN_CONG");
                if (!hasRole) return;

                const inp = userState.taskInputs[t.Task_ID] || { status: "THUC_HIEN", tiendo: 100, chatluong: 100, isX: false };
                
                if (inp.status !== "KHONG_PHAN_CONG") {
                    totalTasks++;
                    totalA += t.Diem_Quy_Doi_Toi_Da;

                    let diemB = 0;
                    if (inp.status === "THUC_HIEN") {
                        const diemTH = t.Diem_Chuan * ((0.3 * (inp.tiendo / 100)) + (0.7 * (inp.chatluong / 100)));
                        diemB = diemTH * t.He_So_Do_Kho;
                    }
                    totalB += diemB;

                    if (inp.isX && inp.status === "THUC_HIEN" && inp.tiendo === 100 && inp.chatluong === 100) {
                        countX++;
                        if (t.Loai_Cong_Viec === "Đột xuất" || t.Diem_Chuan === 12) {
                            const taskKPIScore = (diemB / (totalA || 1)) * 70;
                            bonusFromAdHoc += (0.05 * taskKPIScore);
                        }
                    }
                }
            });

            const scorePartA1 = userState.scoreA1 || 18;
            const scorePartA2 = userState.scoreA2 || 4;
            const scorePartA3 = userState.scoreA3 || 8;
            const scorePartA = scorePartA1 + scorePartA2 + scorePartA3;

            if (document.getElementById("score_partA_total")) {
                document.getElementById("score_partA_total").innerText = `${scorePartA.toFixed(2)} / 30đ`;
            }

            let kpiChuyenMon = totalA > 0 ? (totalB / totalA) * 70 : 0;
            if (kpiChuyenMon > 70) kpiChuyenMon = 70;
            document.getElementById("score_partB").innerText = kpiChuyenMon.toFixed(2);

            let bonusScore = bonusFromAdHoc;
            if (bonusScore > 7) bonusScore = 7;
            document.getElementById("score_partC").innerText = "+" + bonusScore.toFixed(2);

            let finalScore = scorePartA + kpiChuyenMon + bonusScore;
            if (finalScore > 100) finalScore = 100;
            document.getElementById("final_total_score").innerText = finalScore.toFixed(2);

            const ratioX = totalTasks > 0 ? (countX / totalTasks) * 100 : 0;
            document.getElementById("ratio_vuot_muc").innerText = `${ratioX.toFixed(1)}% (${countX}/${totalTasks})`;

            const badge = document.getElementById("badge_xeploai");
            if (finalScore >= 90 && ratioX >= 30) {
                badge.innerText = "Hoàn Thành Xuất Sắc";
                badge.className = "px-2.5 py-1 bg-emerald-500/20 text-emerald-300 border border-emerald-500/30 text-xs font-bold rounded-full";
            } else if (finalScore >= 70) {
                badge.innerText = "Hoàn Thành Tốt";
                badge.className = "px-2.5 py-1 bg-blue-500/20 text-blue-300 border border-blue-500/30 text-xs font-bold rounded-full";
            } else if (finalScore >= 50) {
                badge.innerText = "Hoàn Thành Nhiệm Vụ";
                badge.className = "px-2.5 py-1 bg-amber-500/20 text-amber-300 border border-amber-500/30 text-xs font-bold rounded-full";

        function autoAdjustPoints() {
            const type = document.getElementById("add_task_loai").value;
            const diemChuanInp = document.getElementById("add_task_diemchuan");
            const heSoSel = document.getElementById("add_task_heso");

            if (type === "Đột xuất") {
                diemChuanInp.value = "12";
                heSoSel.value = "1.2";
            } else {
                diemChuanInp.value = "10";
                heSoSel.value = "1.0";
            }
        }

        function addNewTask() {
            const name = document.getElementById("add_task_name").value.trim();
            if (!name) {
                alert("Vui lòng nhập Tên công việc phát sinh!");
                return;
            }

            const truc = parseInt(document.getElementById("add_task_truc").value) || 1;
            const loai = document.getElementById("add_task_loai").value;
            const diemChuan = parseFloat(document.getElementById("add_task_diemchuan").value) || 10;
            const heSo = parseFloat(document.getElementById("add_task_heso").value) || 1.0;
            const minhChung = document.getElementById("add_task_minhchung").value.trim();

            const newId = "PS_" + Date.now().toString().slice(-5);
            const diemQuyDoi = diemChuan * heSo;

            const newTaskObj = {
                "Task_ID": newId,
                "Truc": truc,
                "Task_Name": name,
                "Loai_Cong_Viec": loai,
                "Diem_Chuan": diemChuan,
                "He_So_Do_Kho": heSo,
                "Diem_Quy_Doi_Toi_Da": diemQuyDoi,
                "Required_Roles": ["ALL_GV"],
                "Ghi_Chu_Goc": "Công việc phát sinh cá nhân tự thêm",
                "Minh_Chung_Mac_Dinh": minhChung || "Quyết định / Văn bản phân công hoàn thành trong Quý"
            };

            MASTER_DB.push(newTaskObj);

            userState.taskInputs[newId] = {
                status: "THUC_HIEN",
                tiendo: 100,
                chatluong: 100,
                isX: false,
                minhChung: minhChung || "Quyết định / Văn bản phân công hoàn thành trong Quý"
            };

            // Reset modal inputs
            document.getElementById("add_task_name").value = "";
            document.getElementById("add_task_minhchung").value = "";

            closeAddTaskModal();
            filterAndRender();

            alert(`Đã thêm thành công công việc phát sinh: "${name}"!`);
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
