	\sectionheader{PHẦN I. TRẮC NGHIỆM NHIỀU LỰA CHỌN (10 CÂU)}{Khoanh tròn vào chữ cái A, B, C hoặc D trước phương án trả lời đúng nhất.}
	
	\setcounter{ex}{0}
	\setlength{\columnsep}{18pt}
	\begin{multicols}{2}
		
		\begin{ex}
			Cho góc $\alpha$ thỏa mãn $90^\circ < \alpha < 180^\circ$ và $\sin \alpha + \cos \alpha = \dfrac{1}{5}$. Tính giá trị của biểu thức $P = \tan \alpha - \cot \alpha$.
			\choice
			{\True $P = -\dfrac{7}{12}$}
			{$P = \dfrac{7}{12}$}
			{$P = -\dfrac{35}{12}$}
			{$P = \dfrac{35}{12}$}
		\end{ex}
		
		\begin{ex}
			Cho góc $\alpha$ thỏa mãn $0^\circ < \alpha < 180^\circ, \alpha \neq 90^\circ$ và $\tan \alpha + \cot \alpha = -3$. Tính giá trị của biểu thức $Q = \tan^3 \alpha + \cot^3 \alpha$.
			\choice
			{\True $Q = -18$}
			{$Q = -36$}
			{$Q = 18$}
			{$Q = -27$}
		\end{ex}
		
		\begin{ex}
			Cho góc $\alpha$ thỏa mãn $\tan \alpha = 2$. Tính giá trị của biểu thức $M = \dfrac{3\sin \alpha - 2\cos \alpha}{4\sin \alpha + 5\cos \alpha}$.
			\choice
			{\True $M = \dfrac{4}{13}$}
			{$M = \dfrac{2}{13}$}
			{$M = \dfrac{8}{13}$}
			{$M = -\dfrac{4}{13}$}
		\end{ex}
		
		\begin{ex}
			Cho góc $\alpha$ ($0^\circ \le \alpha \le 180^\circ, \alpha \neq 90^\circ$) thỏa mãn $\sin \alpha \cos \alpha = \dfrac{2}{5}$. Tính giá trị của biểu thức $E = \tan^2 \alpha + \cot^2 \alpha$.
			\choice
			{\True $E = \dfrac{17}{4}$}
			{$E = \dfrac{21}{4}$}
			{$E = \dfrac{25}{4}$}
			{$E = \dfrac{9}{4}$}
		\end{ex}
		
		\begin{ex}
			Tính giá trị của biểu thức $A = \cos^2 10^\circ + \cos^2 20^\circ + \cos^2 30^\circ + \dots + \cos^2 180^\circ$.
			\choice
			{\True $A = 9$}
			{$A = 8$}
			{$A = 8{,}5$}
			{$A = 9{,}5$}
		\end{ex}
		
		\begin{ex}
			Biết $\sin x + \sin(180^\circ - x) = \sqrt{2}$ với $0^\circ \le x \le 180^\circ$. Giá trị của $x$ là:
			\choice
			{\True $x = 45^\circ$ hoặc $x = 135^\circ$}
			{$x = 45^\circ$}
			{$x = 135^\circ$}
			{$x = 60^\circ$ hoặc $x = 120^\circ$}
		\end{ex}
		
		\begin{ex}
			Cho tam giác $ABC$ không vuông. Giá trị của biểu thức $T = \tan A + \tan(B+C)$ bằng:
			\choice
			{\True $T = 0$}
			{$T = 1$}
			{$T = 2\tan A$}
			{$T = -1$}
		\end{ex}
		
		\begin{ex}
			Cho góc $\alpha$ thỏa mãn $\cos \alpha = -\dfrac{2}{3}$. Giá trị của biểu thức $P = 3\sin^2 \alpha + 2\cos^2 \alpha$ bằng:
			\choice
			{\True $P = \dfrac{23}{9}$}
			{$P = \dfrac{19}{9}$}
			{$P = \dfrac{25}{9}$}
			{$P = \dfrac{17}{9}$}
		\end{ex}
		
		\begin{ex}
			Cho $0^\circ < \alpha < 90^\circ$ và $90^\circ < \beta < 180^\circ$. Khẳng định nào sau đây ĐÚNG?
			\choice
			{\True $\sin \alpha \cdot \cos \beta < 0$}
			{$\cos \alpha \cdot \cos \beta > 0$}
			{$\tan \alpha \cdot \tan \beta > 0$}
			{$\sin \alpha \cdot \sin \beta < 0$}
		\end{ex}
		
		\begin{ex}
			Giá trị lớn nhất của biểu thức $F(\alpha) = 2\sin \alpha - 3\cos^2 \alpha + 1$ với $0^\circ \le \alpha \le 180^\circ$ bằng:
			\choice
			{\True $\max F = 3$}
			{$\max F = 4$}
			{$\max F = 2$}
			{$\max F = \dfrac{11}{3}$}
		\end{ex}

	\sectionheader{PHẦN II. TRẮC NGHIỆM ĐÚNG -- SAI (4 CÂU -- MỖI CÂU GỒM 4 Ý a, b, c, d)}{Học sinh xác định mỗi khẳng định là Đúng (Đ) hoặc Sai (S) và đánh dấu vào ô tương ứng.}
	
	\setcounter{ex}{0}
	
	\begin{ex}
		Cho góc $\alpha$ thỏa mãn $90^\circ < \alpha < 180^\circ$ và $\sin \alpha = \dfrac{3}{5}$. Xét tính đúng/sai của các khẳng định sau:
		\choiceTF
		{$\cos \alpha = \dfrac{4}{5}$.}
		{\True $\tan \alpha = -\dfrac{3}{4}$.}
		{Giá trị của biểu thức $P = \dfrac{\tan \alpha + \cot \alpha}{1 + \tan^2 \alpha}$ bằng $-\dfrac{3}{5}$.}
		{$\sin(180^\circ - \alpha) + \cos(180^\circ - \alpha) = -\dfrac{1}{5}$.}
	\end{ex}
	
	\begin{ex}
		Cho hai biểu thức $A(x) = \sin^4 x + \cos^4 x + 2\sin^2 x \cos^2 x$ và $B(x) = \sin^6 x + \cos^6 x + 3\sin^2 x \cos^2 x$ với $0^\circ \le x \le 180^\circ$. Xét tính đúng/sai của các khẳng định sau:
		\choiceTF
		{\True Giá trị của $A(x)$ không phụ thuộc vào biến $x$ và bằng $1$.}
		{\True Biểu thức $B(x)$ rút gọn bằng $1$ với mọi $x$.}
		{\True Với $x = 120^\circ$, giá trị của $A(120^\circ) + B(120^\circ) = 2$.}
		{\True Giá trị lớn nhất của $C(x) = \sin^4 x + \cos^4 x$ trên $[0^\circ, 180^\circ]$ bằng $1$.}
	\end{ex}
	
	\begin{ex}
		Cho góc $\alpha$ thỏa mãn $\tan \alpha + \cot \alpha = 3$. Xét tính đúng/sai của các khẳng định sau:
		\choiceTF
		{\True Cả $\sin \alpha$ và $\cos \alpha$ bắt buộc phải cùng dấu.}
		{\True $\tan^2 \alpha + \cot^2 \alpha = 7$.}
		{\True $|\tan \alpha - \cot \alpha| = \sqrt{5}$.}
		{Nếu $0^\circ < \alpha < 45^\circ$ thì $\tan^3 \alpha - \cot^3 \alpha = 7\sqrt{5}$.}
	\end{ex}
	
	\begin{ex}
		Một chất điểm chịu tác dụng của hai lực $\vec{F}_1$ và $\vec{F}_2$ có độ lớn lần lượt là $F_1 = 10\text{ N}$ và $F_2 = 10\sqrt{3}\text{ N}$. Góc giữa hai lực $\vec{F}_1$ và $\vec{F}_2$ là $\theta$ ($0^\circ \le \theta \le 180^\circ$). Xét tính đúng/sai của các khẳng định sau:
		\choiceTF
		{\True Nếu $\theta = 90^\circ$, độ lớn lực tổng hợp $\vec{F} = \vec{F}_1 + \vec{F}_2$ bằng $20\text{ N}$.}
		{Nếu $\theta = 120^\circ$, độ lớn lực tổng hợp $F = 10\text{ N}$.}
		{\True Độ lớn lực tổng hợp $F$ đạt giá trị lớn nhất bằng $10(1+\sqrt{3})\text{ N}$ khi $\theta = 0^\circ$.}
		{Khi $\theta = 150^\circ$, công sinh ra bởi lực tổng hợp $\vec{F}$ làm dịch chuyển vật một đoạn $d = 5\text{ m}$ theo hướng của $\vec{F}_1$ là $25\text{ J}$.}
	\end{ex}

	\sectionheader{PHẦN III. CÂU HỎI TRẢ LỜI NGẮN (5 CÂU)}{Học sinh viết đáp số hoặc câu trả lời ngắn gọn vào ô trống tương ứng.}
	
	\setcounter{ex}{0}
	
	\begin{ex}
		Một cabin của vòng quay mặt trời bán kính $R = 50\text{ m}$ quay đều quanh tâm $O$. Tại thời điểm ban đầu, cabin ở vị trí thấp nhất $A$. Sau một thời gian quay, góc quay của bán kính nối cabin $\widehat{AOM} = \alpha = 150^\circ$. Hỏi độ cao của cabin ở vị trí $M$ so với vị trí thấp nhất $A$ bằng bao nhiêu mét (làm tròn kết quả đến hàng phần mười)?
		\shortans{93,3 m}
	\end{ex}
	
	\begin{ex}
		Một cánh tay robot dài $1{,}2\text{ m}$ gắp vật từ vị trí điểm $A$ nằm ngang bên phải tâm quay $O$. Robot quay một góc $\alpha = 135^\circ$ quanh tâm $O$ đến vị trí điểm $B$ rồi thả vật. Khoảng cách giữa vị trí đầu $A$ và vị trí thả $B$ bằng bao nhiêu mét (làm tròn kết quả đến hàng phần trăm)?
		\shortans{2,22 m}
	\end{ex}
	
	\begin{ex}
		Cho góc $\alpha$ thỏa mãn $90^\circ < \alpha < 180^\circ$ và $\tan \alpha + \cot \alpha = -\dfrac{25}{12}$. Tính giá trị của biểu thức $\sin \alpha + \cos \alpha$.
		\shortans{-0,2}
	\end{ex}
	
	\begin{ex}
		Cho góc $\alpha$ thỏa mãn $\sin \alpha = \dfrac{1}{3}$. Tính giá trị của biểu thức $P = \dfrac{\sin(180^\circ - \alpha) + 2\cos(90^\circ - \alpha)}{\tan(180^\circ - \alpha) \cdot \cot \alpha}$.
		\shortans{-1}
	\end{ex}
	
	\begin{ex}
		Tính giá trị của biểu thức $S = \sin^2 10^\circ + \sin^2 20^\circ + \sin^2 30^\circ + \dots + \sin^2 180^\circ$.
		\shortans{9}
	\end{ex}
